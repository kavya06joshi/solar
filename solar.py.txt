"""
Dual-axis solar tracker v2 - Blender model WITH the real mechanisms
-------------------------------------------------------------------
HOW TO RUN
  1. Open Blender, go to the "Scripting" workspace.
  2. Text > Open > pick this file, then press "Run Script".
  3. Press SPACE in the 3D viewport to play the sun-path animation.

WHAT IS NEW COMPARED TO v1
  * Poles are now 2-stage TELESCOPIC LINEAR ACTUATORS (base housing + two sliding
    tubes + motor + top clevis). The tubes really slide in and out.
  * The rotating mat sits on a SLEW DRIVE: fixed foundation + pedestal, a bearing,
    a toothed slew ring (turns with the mat) and a WORM GEAR with motor that is
    fixed to the ground. The worm spins while the ring turns (48 teeth).
  * Wheels now roll on two GUIDE RAILS on the mat.
  * Controller box on a post (cable to the foundation) and a WIND SENSOR
    (spinning cups) on a mast - this is what triggers the "stow flat" mode.
  * Three cameras: Cam_Main (active), Cam_Slew, Cam_Actuator.
    To switch: click a camera in the Outliner, then View > Cameras > Set Active Object as Camera
    (or Ctrl+Numpad0).

The console prints how long the actuators must be over a FULL YEAR, because the
sun is much higher in summer than on the day shown in the animation.
"""

def clear_scene():
    if bpy.context.object and bpy.context.object.mode != 'OBJECT':
        bpy.ops.object.mode_set(mode='OBJECT')
    bpy.ops.object.select_all(action='SELECT')
    bpy.ops.object.delete(use_global=False)


def make_mat(name, rgb, metallic=0.0, rough=0.5, emit=False):
    m = bpy.data.materials.new(name)
    m.diffuse_color = (*rgb, 1.0)
    try:
        m.use_nodes = True
        bsdf = m.node_tree.nodes.get("Principled BSDF")
        if bsdf:
            bsdf.inputs["Base Color"].default_value = (*rgb, 1.0)
            bsdf.inputs["Metallic"].default_value = metallic
            bsdf.inputs["Roughness"].default_value = rough
            if emit:
                for key in ("Emission Color", "Emission"):
                    if key in bsdf.inputs:
                        bsdf.inputs[key].default_value = (*rgb, 1.0)
                        break
                if "Emission Strength" in bsdf.inputs:
                    bsdf.inputs["Emission Strength"].default_value = 1.0
    except Exception:
        pass
    return m


def assign(obj, mat):
    obj.data.materials.clear()
    obj.data.materials.append(mat)


def add_box(name, dims, loc, mat, parent=None):
    bpy.ops.mesh.primitive_cube_add(size=1, location=loc)
    o = bpy.context.active_object
    o.name = name
    o.scale = dims
    assign(o, mat)
    if parent:
        o.parent = parent
    return o


def add_cyl(name, radius, depth, loc, mat, parent=None, bake=None, base_at_zero=False, verts=32):
    bpy.ops.mesh.primitive_cylinder_add(radius=radius, depth=depth, location=loc, vertices=verts)
    o = bpy.context.active_object
    o.name = name
    if bake is not None:
        o.data.transform(bake)
    if base_at_zero:
        o.data.transform(Matrix.Translation((0, 0, depth / 2)))
    assign(o, mat)
    if parent:
        o.parent = parent
    return o


def add_sphere(name, radius, loc, mat, parent=None):
    bpy.ops.mesh.primitive_uv_sphere_add(radius=radius, location=loc)
    o = bpy.context.active_object
    o.name = name
    assign(o, mat)
    if parent:
        o.parent = parent
    return o


def add_rod(name, p1, p2, radius, mat, parent=None):
    p1, p2 = Vector(p1), Vector(p2)
    d = p2 - p1
    bpy.ops.mesh.primitive_cylinder_add(radius=radius, depth=d.length,
                                        location=(p1 + p2) / 2, vertices=12)
    o = bpy.context.active_object
    o.name = name
    o.rotation_mode = 'QUATERNION'
    o.rotation_quaternion = d.to_track_quat('Z', 'Y')
    assign(o, mat)
    if parent:
        o.parent = parent
    return o


def make_camera(name, loc, target_loc, lens=50):
    cd = bpy.data.cameras.new(name)
    cd.lens = lens
    cd.clip_end = 200
    c = bpy.data.objects.new(name, cd)
    bpy.context.collection.objects.link(c)
    c.location = loc
    bpy.ops.object.empty_add(type='PLAIN_AXES', location=target_loc)
    t = bpy.context.active_object
    t.name = name + "_Target"
    tc = c.constraints.new('TRACK_TO')
    tc.target = t
    tc.track_axis = 'TRACK_NEGATIVE_Z'
    tc.up_axis = 'UP_Y'
    return c


def add_label(text, loc, mat, cam, parent=None, size=0.2, align='CENTER'):
    bpy.ops.object.text_add(location=loc)
    o = bpy.context.active_object
    o.name = "Label_" + text[:18]
    o.data.body = text
    o.data.size = size
    o.data.align_x = align
    assign(o, mat)
    c = o.constraints.new('TRACK_TO')
    c.target = LABEL_TARGET          # an empty that always follows the ACTIVE camera
    c.track_axis = 'TRACK_Z'
    c.up_axis = 'UP_Y'
    if parent:
        o.parent = parent
    return o


# ---------- sun model ----------
def sun_position(hour, doy=None):
    """Return (azimuth from North clockwise, elevation) in degrees for solar time."""
    doy = DAY_OF_YEAR if doy is None else doy
    lat = math.radians(LATITUDE)
    decl = math.radians(23.45 * math.sin(math.radians(360.0 / 365.0 * (284 + doy))))
    H = math.radians(15.0 * (hour - 12.0))
    sin_el = math.sin(lat) * math.sin(decl) + math.cos(lat) * math.cos(decl) * math.cos(H)
    el = math.asin(max(-1, min(1, sin_el)))
    cos_az = (math.sin(decl) - math.sin(el) * math.sin(lat)) / (math.cos(el) * math.cos(lat))
    az = math.acos(max(-1, min(1, cos_az)))
    if H > 0:
        az = 2 * math.pi - az
    return math.degrees(az), math.degrees(el)


def sun_vector(az, el):
    a, e = math.radians(az), math.radians(el)
    return Vector((math.sin(a) * math.cos(e), math.cos(a) * math.cos(e), math.sin(e)))


def solve_pose(hour, doy=None):
    az, el = sun_position(hour, doy)
    tilt = max(MIN_TILT, min(MAX_TILT, 90.0 - el))
    yaw = 180.0 - az
    th = math.radians(tilt)
    z_top = Z_B + PANEL_L * math.sin(th)
    y_bot = Y_POLE - PANEL_L * math.cos(th)
    return dict(az=az, el=el, tilt=tilt, yaw=yaw, th=th, z_top=z_top, y_bot=y_bot,
                pole_h=z_top - MAT_TOP, wheel_dist=PANEL_L * math.cos(th))


def annual_ranges():
    """Pole height and wheel position extremes over a whole year (daylight only)."""
    hs, ys = [], []
    for doy in range(1, 366, 5):
        for q in range(20, 80):
            hour = q / 4.0
            az, el = sun_position(hour, doy)
            if el < 3:
                continue
            p = solve_pose(hour, doy)
            hs.append(p["pole_h"])
            ys.append(p["y_bot"])
    return min(hs), max(hs), min(ys), max(ys)

# ============================== POSE + ANIMATION ==============================
def key(obj, prop, frame):
    obj.keyframe_insert(prop, frame=frame)


def apply_pose(pose, frame):
    DIST = 14.0
    turntable.rotation_euler = (0, 0, math.radians(pose["yaw"]))
    key(turntable, "rotation_euler", frame)

    # worm spins SLEW_TEETH times faster than the ring
    worm.rotation_euler = (0, math.radians(pose["yaw"]) * SLEW_TEETH, 0)
    key(worm, "rotation_euler", frame)

    # actuators: two telescopic stages share the extension equally
    ext = max(0.0, pose["pole_h"] - ACT_BASE_H)
    seg = 0.2 + ext / 2.0
    for s1, s2, cl, px in zip(stage1, stage2, clevis, POLE_X):
        s1.scale = (1, 1, seg)
        key(s1, "scale", frame)
